<script>
  var total = 15;
  var sec = total;
  var secEl = document.getElementById('sec');
  var timer;

  function endSim(){
    clearInterval(timer);
    document.getElementById('content').style.display = 'none';
    document.getElementById('flash').style.display = 'none';
    document.getElementById('end').style.display = 'flex';
    window.scrollTo(0,0);
  }

  timer = setInterval(function(){
    sec--;
    if(secEl) secEl.textContent = sec;
    if(sec <= 0) endSim();
  }, 1000);

  document.getElementById('stopBtn').addEventListener('click', endSim);

  // 溫和震動（可選，不支援就忽略）
  if(navigator.vibrate){
    var v = setInterval(function(){ navigator.vibrate(60); }, 1000);
    setTimeout(function(){ clearInterval(v); navigator.vibrate(0); }, total*1000);
  }
